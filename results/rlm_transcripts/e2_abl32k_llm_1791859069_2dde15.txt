    p = ("Classify each question by the type of its answer into exactly one of these 6 categories (TREC-style):\n"
         "- location (cities, countries, mountains, etc.)\n"
         "- human being (a person, group, or individual)\n"
         "- entity (animals, bodies, colors, creative works, currency, diseases, food, instruments, products, religion, sports, substances, symbols, techniques, terms, vehicles, words, other things)\n"
         "- abbreviation (an abbreviation or what an abbreviation expands to)\n"
         "- numeric value (counts, dates, distances, money, percent, period, speed, temperature, size, weight, codes)\n"
         "- description and abstract concept (definitions, descriptions, manner, reasons)\n\n"
         "Output one line per question in the format '<number>: <category>' using exactly the category names above. No other text.\n\n")
    for i, q in enumerate(chunk):
        p += f"{start+i}: {q}\n"
    return p

size = 50
